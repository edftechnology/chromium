\documentclass[msc]{ita}    % ITA.cls based on standard book.cls

\input{preamble.tex}
\input{variables.tex}

\frontmatter
\makenoidxglossaries
\input{list_of_acronyms.tex}
\input{list_of_symbols.tex}

%\hyphenation{param-e-ter-ization}
 
\begin{document}
	
	\thispagestyle{empty}
	\input{cover_page} \clearpage  % [x] Revisão do texto
	
	% Folha de Rosto e Capa para o caso do TG
	\maketitle
	
	% Dedicatoria:
	\input{dedication.tex}  % [x] Revisão do texto
	
	% Agradecimentos:
	\input{acknowledgments.tex}  % [x] Revisão do texto
	
	% Epigrafo:
	\input{epigraph.tex}  % [x] Revisão do texto
	
	% Resumo
	\input{resumo.tex}  % [x] Revisão do texto
	
	% Abstract
	\input{abstract.tex}  % [x] Revisão do texto
	
	% Listas:
	\glsaddall

	% Lista de abreviaturas
	\printnoidxglossary[type=\acronymtype,
	    title=\listofabbreviationsname,
	    sort=standard,
	    nonumberlist]
	\pdfbookmark[0]{\listofabbreviationsname}{listaabreviaturas}

	% Lista de simbolos
	\printnoidxglossary[
	    title=\listofsymbolsname,
	    sort=use,
	    nonumberlist]
	\pdfbookmark[0]{\listofsymbolsname}{listasimbolos}

	\input{list_of_definitions.tex}
	\input{list_of_illustrations.tex}
	\input{list_of_nomenclatures.tex}
	\input{list_of_tables.tex}
	
	% Sumario
	\tableofcontents
	
	\mainmatter
	% Os capitulos comecam aqui
	
	% Desenvolvimento:
	\input{introduction.tex}
	\input{contextualization.tex}
	
	% Descrição numérica:
	\input{numerical_description.tex}
	\input{resultados_e_discussoes.tex}
	
	% Conclusão e trabalhos futuros:
%	\input{conclusion.tex}
%	\input{future_works.tex}
%	\input{outras_aplicacoes_em_engenharia.tex}
	
	% Apêndices:
    \input{appendix.tex}
	
	% Anexos:
	\input{attachments.tex}
	
	% REFERENCIAS BIBLIOGRAFICAS
%	\input{references.tex}
	\renewcommand\bibname{\itareferencesnamebabel}
	\bibliography{references}	
		
	\input{document_registration_sheet.tex}
	
\end{document}
